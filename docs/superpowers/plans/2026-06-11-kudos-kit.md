# kudos-kit Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a self-hosted cross-product feedback widget system: a vanilla JS snippet (`kkw.js`), a NestJS ingest API, a Postgres database, and a Next.js admin page — all co-located on `feedback.agileturtles.com`.

**Architecture:** Vanilla `kkw.js` POSTs feedback to `POST /v1/feedback` (NestJS) which validates, inserts into Postgres, and fires an async Brevo email notification. The Next.js admin page reads via a least-privilege `feedback_admin` DB role and is guarded by a single-password session cookie.

**Tech Stack:** NestJS 11 (server), Next.js 16 App Router (admin), Postgres (DB), esbuild (widget bundler), Brevo REST API (notifications), Cloudflare Turnstile (spam), Node.js crypto (session signing)

> **Note on external scripts:** `kkw.js` dynamically loads `challenges.cloudflare.com/turnstile/v0/api.js`. SRI (`integrity=`) cannot be added to this script — Cloudflare updates it without versioning and their own docs don't recommend SRI for it. If Cloudflare's CDN were compromised, the server-side `siteverify` call would also be void. The right mitigation is a strict CSP on host pages: `script-src 'self' https://challenges.cloudflare.com`.

// server/src/database/database.service.ts
import { Injectable, OnModuleDestroy, OnModuleInit } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { Pool, QueryResult } from 'pg';

@Injectable()
export class DatabaseService implements OnModuleInit, OnModuleDestroy {
  private pool: Pool;

  constructor(private config: ConfigService) {}

  onModuleInit(): void {
    this.pool = new Pool({
      host: this.config.get('database.host'),
      port: this.config.get('database.port'),
      database: this.config.get('database.name'),
      user: this.config.get('database.user'),
      password: this.config.get('database.password'),
    });
  }

  async onModuleDestroy(): Promise<void> {
    await this.pool?.end();
  }

  async query<T>(sql: string, params?: unknown[]): Promise<QueryResult<T>> {
    return this.pool.query<T>(sql, params);
  }
}
```

// server/src/feedback/turnstile.service.ts
import { Injectable } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';

@Injectable()
export class TurnstileService {
  constructor(private config: ConfigService) {}

  async verify(token: string, ip?: string): Promise<boolean> {
    const secret = this.config.get<string>('turnstile.secret') ?? '';
    const body = new URLSearchParams({ secret, response: token });
    if (ip) body.set('remoteip', ip);

    const res = await fetch('https://challenges.cloudflare.com/turnstile/v0/siteverify', {
      method: 'POST',
      body,
    });
    const data = await res.json() as { success: boolean };
    return data.success === true;
  }
}
```

// server/src/feedback/notification.service.ts
import { Injectable } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';

interface NotifyPayload {
  project: string;
  type: string;
  message: string;
  id: string;
}

@Injectable()
export class NotificationService {
  private lastSentAt = 0;
  private readonly throttleMs = 60_000;

  constructor(private config: ConfigService) {}

  async notify(payload: NotifyPayload): Promise<void> {
    if (Date.now() - this.lastSentAt < this.throttleMs) return;
    this.lastSentAt = Date.now();

    const apiKey = this.config.get<string>('brevo.apiKey') ?? '';
    const fromEmail = this.config.get<string>('brevo.fromEmail') ?? '';
    const toEmail = this.config.get<string>('brevo.toEmail') ?? '';
    const adminUrl = this.config.get<string>('adminUrl') ?? '';

    await fetch('https://api.brevo.com/v3/smtp/email', {
      method: 'POST',
      headers: { 'api-key': apiKey, 'Content-Type': 'application/json' },
      body: JSON.stringify({
        sender: { email: fromEmail },
        to: [{ email: toEmail }],
        subject: `[${payload.project}] New ${payload.type} report`,
        textContent: [
          payload.message.slice(0, 200),
          '',
          `${adminUrl}/admin`,
        ].join('\n'),
      }),
    });
  }
}

// server/src/feedback/feedback.service.ts
import { Injectable, BadRequestException } from '@nestjs/common';
import { DatabaseService } from '../database/database.service';
import { TurnstileService } from './turnstile.service';
import { NotificationService } from './notification.service';
import { CreateFeedbackDto } from './dto/create-feedback.dto';

@Injectable()
export class FeedbackService {
  constructor(
    private db: DatabaseService,
    private turnstile: TurnstileService,
    private notification: NotificationService,
  ) {}

  async submit(dto: CreateFeedbackDto, ip: string): Promise<void> {
    if (dto.hp) return;

    const proj = await this.db.query<{ active: boolean }>(
      'SELECT active FROM projects WHERE key = $1',
      [dto.project],
    );
    if (!proj.rows.length || !proj.rows[0].active) {
      throw new BadRequestException('Invalid project');
    }

    const valid = await this.turnstile.verify(dto.turnstileToken, ip);
    if (!valid) throw new BadRequestException('CAPTCHA verification failed');

    const result = await this.db.query<{ id: string }>(
      `INSERT INTO feedback
         (project, type, message, email, user_ctx, context, console_logs, errors, ip)
       VALUES ($1,$2,$3,$4,$5,$6,$7,$8,$9::inet)
       RETURNING id`,
      [
        dto.project,
        dto.type,
        dto.message,
        dto.email ?? null,
        dto.user ? JSON.stringify(dto.user) : null,
        dto.context ? JSON.stringify(dto.context) : null,
        dto.console ? JSON.stringify(dto.console) : null,
        dto.errors ? JSON.stringify(dto.errors) : null,
        ip,
      ],
    );

    this.notification.notify({
      project: dto.project,
      type: dto.type,
      message: dto.message,
      id: result.rows[0].id,
    }).catch(() => {});
  }
}
```

// widget/src/serializer.ts
const REDACT_PATTERN = /authorization|bearer\s+|password|secret|api[_-]?key|token|set-cookie/i;

function safeStringify(value: unknown, depth = 0): string {
  if (depth > 4) return '[truncated]';
  if (value === null) return 'null';
  if (value === undefined) return 'undefined';
  if (typeof value === 'string') return value.slice(0, 2048);
  if (typeof value === 'number' || typeof value === 'boolean') return String(value);
  if (value instanceof Error) return value.stack ?? value.message;
  if (typeof Element !== 'undefined' && value instanceof Element) {
    return `<${value.tagName.toLowerCase()}>`;
  }
  if (typeof value === 'function') return '[Function]';
  if (Array.isArray(value)) {
    return '[' + value.slice(0, 20).map(v => safeStringify(v, depth + 1)).join(',') + ']';
  }
  if (typeof value === 'object') {
    const seen = new Set<unknown>();
    try {
      return JSON.stringify(value, (_key, val) => {
        if (typeof val === 'object' && val !== null) {
          if (seen.has(val)) return '[circular]';
          seen.add(val);
        }
        return val;
      });
    } catch {
      return '[unserializable]';
    }
  }
  return String(value);
}

export function serializeArgs(args: unknown[]): string {
  return args.map(a => safeStringify(a)).join(' ').slice(0, 2048);
}

export function redactString(value: string): string {
  return value.replace(
    /("(?:authorization|password|secret|api[_-]?key|token|set-cookie)":\s*)"([^"]*)"/gi,
    (_, key) => `${key}"[redacted]"`,
  );
}

// widget/src/console-capture.ts
import { ConsoleEntry, ErrorEntry } from './types';
import { serializeArgs, redactString } from './serializer';

interface CaptureConfig {
  max: number;
  maxBytes: number;
  redact: boolean;
}

export class ConsoleCapture {
  private logs: ConsoleEntry[] = [];
  private errors: ErrorEntry[] = [];
  private originals: Record<string, (...args: unknown[]) => void> = {};
  private errorHandler: ((e: ErrorEvent) => void) | null = null;
  private rejectionHandler: ((e: PromiseRejectionEvent) => void) | null = null;

  constructor(private config: CaptureConfig) {}

  install(): void {
    const levels = ['log', 'info', 'warn', 'error', 'debug'] as const;
    for (const level of levels) {
      this.originals[level] = console[level].bind(console);
      console[level] = (...args: unknown[]) => {
        this.originals[level](...args);
        this.push({ t: Date.now(), level, msg: this.serialize(args) });
      };
    }

    this.errorHandler = (e: ErrorEvent) => {
      this.errors.push({
        message: e.message,
        filename: e.filename,
        lineno: e.lineno,
        stack: (e.error as Error | null)?.stack,
      });
    };
    this.rejectionHandler = (e: PromiseRejectionEvent) => {
      const reason = e.reason as Error | string | null;
      this.errors.push({
        message: reason instanceof Error ? reason.message : String(reason ?? 'Unhandled rejection'),
        stack: reason instanceof Error ? reason.stack : undefined,
      });
    };

    window.addEventListener('error', this.errorHandler);
    window.addEventListener('unhandledrejection', this.rejectionHandler);
  }

  restore(): void {
    for (const [level, fn] of Object.entries(this.originals)) {
      (console as any)[level] = fn;
    }
    if (this.errorHandler) window.removeEventListener('error', this.errorHandler);
    if (this.rejectionHandler) window.removeEventListener('unhandledrejection', this.rejectionHandler);
    this.originals = {};
  }

  getLogs(): ConsoleEntry[] {
    return [...this.logs];
  }

  getErrors(): ErrorEntry[] {
    return [...this.errors];
  }

  clear(): void {
    this.logs = [];
    this.errors = [];
  }

  private push(entry: ConsoleEntry): void {
    this.logs.push(entry);
    if (this.logs.length > this.config.max) this.logs.shift();
    this.enforceBytesCap();
  }

  private enforceBytesCap(): void {
    let total = this.logs.reduce((sum, e) => sum + e.msg.length, 0);
    while (total > this.config.maxBytes && this.logs.length > 0) {
      const removed = this.logs.shift()!;
      total -= removed.msg.length;
    }
  }

  private serialize(args: unknown[]): string {
    const raw = serializeArgs(args);
    return this.config.redact ? redactString(raw) : raw;
  }
}

  if (!ui) return;
  ui.hideButton();
  ui.hidePanel();
  emit('visibilitychange', false);
}

function toggle(): void { isVisible() ? hide() : show(); }

function destroy(): void {
  ui?.destroy(); ui = null;
  capture?.restore(); capture = null;
  config = null; listeners = {};
}

async function submitFeedback(): Promise<void> {
  if (!ui || !config) return;
  const message = ui.getMessage();
  if (!message) { ui.showMessage('Please enter a message.', true); return; }

  ui.setSubmitState(true);

  let turnstileToken = '';
  try {
    turnstileToken = await new Promise<string>((resolve, reject) => {
      const container = ui!.getTurnstileContainer();
      if (!container) return reject(new Error('No Turnstile container'));
      (window as any).turnstile.render(container, {
        sitekey: config!.turnstileSiteKey,
        callback: resolve,
        'error-callback': reject,
      });
    });
  } catch {
    ui.showMessage('CAPTCHA failed. Please try again.', true);
    ui.setSubmitState(false);
    return;
  }

  const type = ui.getSelectedType();
  const payload: FeedbackPayload = {
    project: config.project,
    type,
    message,
    email: ui.getEmail() || undefined,
    user: config.user,
    context: {
      url: location.href,
      referrer: document.referrer,
      userAgent: navigator.userAgent,
      viewport: { w: window.innerWidth, h: window.innerHeight },
      language: navigator.language,
      build: (window as any).__APP_VERSION__,
      ts: new Date().toISOString(),
    },
    hp: ui.getHoneypot(),
    turnstileToken,
  };

  const consoleCfg = config.console;
  if (capture && consoleCfg?.sendOn.includes(type)) {
    payload.console = capture.getLogs();
    payload.errors = capture.getErrors();
  }

  try {
    const res = await fetch(INGEST_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
    });
    if (!res.ok) throw new Error(`${res.status}`);
    ui.showMessage('Thanks for your feedback!');
    ui.setSubmitState(false);
    capture?.clear();
  } catch {
    ui.showMessage('Failed to send. Please try again.', true);
    ui.setSubmitState(false);
  }
}

export function init(cfg: FeedbackConfig): void {
  if (ui) return;
  config = cfg;

  if (cfg.console?.enabled) {
    capture = new ConsoleCapture({
      max: cfg.console.max,
      maxBytes: cfg.console.maxBytes,
      redact: cfg.console.redact,
    });
    capture.install();
  }

  ui = new FeedbackUI(cfg);
  ui.mount();

  const submitBtn = document.querySelector<HTMLButtonElement>('.kkw-submit');
  submitBtn?.addEventListener('click', submitFeedback);

  if (cfg.startHidden) hide();

  // Load Turnstile script if not already present
  if (!(window as any).turnstile) {
    const script = document.createElement('script');
    script.src = 'https://challenges.cloudflare.com/turnstile/v0/api.js';
    script.async = true;
    document.head.appendChild(script);
  }
}

// client/src/lib/auth.ts
'server only'

import { createHmac, timingSafeEqual } from 'crypto';
import { cookies } from 'next/headers';
import { redirect } from 'next/navigation';

const SECRET = process.env.SESSION_SECRET!;
const MAX_AGE_S = 7 * 24 * 60 * 60;

function sign(value: string): string {
  return createHmac('sha256', SECRET).update(value).digest('hex');
}

export async function createSession(): Promise<void> {
  const expiry = Math.floor(Date.now() / 1000) + MAX_AGE_S;
  const token = `${expiry}:${sign(String(expiry))}`;
  const store = await cookies();
  store.set('kkw_session', token, {
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'lax',
    maxAge: MAX_AGE_S,
    path: '/',
  });
}

export async function getSession(): Promise<boolean> {
  const store = await cookies();
  const token = store.get('kkw_session')?.value;
  if (!token) return false;
  const colonIdx = token.indexOf(':');
  if (colonIdx === -1) return false;
  const expiry = token.slice(0, colonIdx);
  const sig = token.slice(colonIdx + 1);
  const expected = sign(expiry);
  try {
    const sigBuf = Buffer.from(sig, 'hex');
    const expBuf = Buffer.from(expected, 'hex');
    if (sigBuf.length !== expBuf.length) return false;
    if (!timingSafeEqual(sigBuf, expBuf)) return false;
  } catch {
    return false;
  }
  return parseInt(expiry, 10) > Math.floor(Date.now() / 1000);
}

export async function requireAuth(): Promise<void> {
  if (!(await getSession())) redirect('/login');
}

export async function deleteSession(): Promise<void> {
  (await cookies()).delete('kkw_session');
}
```
